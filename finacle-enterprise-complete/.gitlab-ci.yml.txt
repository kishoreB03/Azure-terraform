image:
  name: hashicorp/terraform:1.8

stages:
  - validate
  - plan
  - approval
  - apply

before_script:
  - terraform --version

validate:
  stage: validate
  script:
    - terraform init
    - terraform fmt -check
    - terraform validate

plan:
  stage: plan
  script:
    - terraform init
    - terraform plan -out=tfplan
  artifacts:
    paths:
      - tfplan

approval:
  stage: approval
  script:
    - echo "Awaiting approval"
  when: manual
  allow_failure: false

apply:
  stage: apply
  script:
    - terraform apply -auto-approve tfplan
  dependencies:
    - plan
  when: manual
